## 📋 核心需求

### 研发提供的静态端口列表
```python
STATIC_PORTS_PROVIDED = [
    22, 53, 1701, 9001, 68, 55832, 40404,
    500, 4500, 123, 1, 58, 7574, 9993, 520
]
# 共15个端口
```

### 验证目标
**通过遍历1-65535所有端口，验证静态列表的完整性**

### 验证逻辑
```
1. 在路由器的某个端口配置页面
2. 依次填写端口1, 2, 3, ..., 65535（共65535个端口）
3. 每次点击保存/应用，检测是否弹窗提示冲突
4. 记录所有提示冲突的端口号
5. 对比记录结果和研发提供的列表

判断标准:
  ✅ 如果所有冲突端口 == 研发列表 → 测试通过（列表正确）
  ❌ 如果有冲突端口不在列表中 → 测试失败（列表有遗漏）
  ❌ 如果列表中的端口未冲突 → 测试失败（列表有多余）
```

#### 测试步骤
```python
1. 准备阶段:
   - 登录路由器
   - 进入某个支持端口配置的页面（需要确认具体页面）
   - 准备研发提供的静态端口列表

2. 遍历测试:
   for port in range(1, 65536):  # 1到65535
       a. 在端口输入框输入: port
       b. 点击保存/应用按钮
       c. 检测是否弹出冲突提示弹窗
       d. 如果弹窗:
          - 记录端口号到 actual_conflict_ports
          - 关闭弹窗
       e. 如果无弹窗:
          - 记录端口号到 available_ports
          - 恢复原始端口（避免真的改变配置）

3. 结果对比:
   - 将 actual_conflict_ports 与 STATIC_PORTS_PROVIDED 对比

4. 生成报告:
   - 列表中的端口是否都冲突了？
   - 是否有遗漏的冲突端口？
   - 是否有多余的端口（列表中但不冲突）？
```

#### 预期结果
```
✅ 完美情况:
   actual_conflict_ports == STATIC_PORTS_PROVIDED
   → 测试通过，静态列表完全正确

❌ 有遗漏:
   actual_conflict_ports = [22, 53, 80, 443, ...]  # 实际冲突
   STATIC_PORTS_PROVIDED = [22, 53, ...]           # 研发提供
   遗漏端口 = [80, 443, ...]  # 实际冲突但列表没有

❌ 有多余:
   actual_conflict_ports = [22, 53, ...]           # 实际冲突
   STATIC_PORTS_PROVIDED = [22, 53, 80, 443, ...]  # 研发提供
   多余端口 = [80, 443, ...]  # 列表有但实际不冲突
   → 测试失败，提示："静态端口列表有多余，端口80, 443实际未占用"
```

## 🔧 测试实现代码框架

```python
class StaticPortListVerificationTest(BaseTest):
    """静态端口列表完整性验证测试"""

    @property
    def test_name(self) -> str:
        return "静态端口列表完整性验证"

    @property
    def description(self) -> str:
        return "遍历1-65535所有端口，验证研发提供的静态端口列表是否完整准确"

    def __init__(self, config: TestConfig = None):
        super().__init__(config)

        # 研发提供的静态端口列表
        self.STATIC_PORTS_PROVIDED = [
            22, 53, 1701, 9001, 68, 55832, 40404,
            500, 4500, 123, 1, 58, 7574, 9993, 520
        ]

        # 实际检测到的冲突端口
        self.actual_conflict_ports = []

        # 可用端口（无冲突）
        self.available_ports = []

        # 测试页面配置（待确认）
        self.test_page_url = "/#xxx/xxx"  # 需要确认
        self.port_input_xpath = '//*[@id="xxx_port"]'  # 需要确认
        self.save_button_xpath = '//*[@id="xxx_save"]'  # 需要确认
        self.popup_xpath = None  # 待确认弹窗类型

    def setup(self):
        """测试前置条件"""
        self.log_info("="*70)
        self.log_info("开始静态端口列表完整性验证测试")
        self.log_info("="*70)

        # 登录路由器
        self.router_client.login_web()

        # 进入测试页面
        self.router_client.driver.get(
            f"http://{self.router_client.router_ip}{self.test_page_url}"
        )
        time.sleep(2)

        self.log_info(f"研发提供的静态端口列表（共{len(self.STATIC_PORTS_PROVIDED)}个）:")
        self.log_info(f"  {sorted(self.STATIC_PORTS_PROVIDED)}")
        self.log_info(f"\n准备遍历端口1-65535进行验证...\n")

    def execute(self):
        """执行测试：遍历所有端口"""
        total_ports = 65535
        check_interval = 1000  # 每1000个端口输出一次进度

        for port in range(1, total_ports + 1):
            # 输出进度
            if port % check_interval == 0:
                progress = (port / total_ports) * 100
                self.log_info(f"进度: {port}/{total_ports} ({progress:.1f}%) - "
                            f"已发现冲突端口: {len(self.actual_conflict_ports)}个")

            # 测试端口
            is_conflict = self._test_single_port(port)

            if is_conflict:
                self.actual_conflict_ports.append(port)
                # 重要端口立即输出
                if port in self.STATIC_PORTS_PROVIDED or port < 1024:
                    self.log_info(f"  ⚠️ 端口 {port} 冲突")
            else:
                self.available_ports.append(port)

        # 输出最终进度
        self.log_info(f"\n✅ 遍历完成: {total_ports}/{total_ports} (100%)")
        self.log_info(f"   发现冲突端口: {len(self.actual_conflict_ports)}个")
        self.log_info(f"   可用端口: {len(self.available_ports)}个\n")

        # 对比结果
        self._compare_results()

    def _test_single_port(self, port: int) -> bool:
        """
        测试单个端口是否冲突

        Returns:
            bool: True=冲突, False=可用
        """
        try:
            # 1. 找到端口输入框
            port_input = self.router_client.driver.find_element(
                By.XPATH, self.port_input_xpath
            )

            # 2. 清空并输入端口
            port_input.clear()
            port_input.send_keys(str(port))

            # 3. 点击保存按钮
            save_button = self.router_client.driver.find_element(
                By.XPATH, self.save_button_xpath
            )
            save_button.click()

            # 4. 等待并检测弹窗
            time.sleep(0.5)  # 等待弹窗出现

            # 5. 检测是否有冲突弹窗（具体实现取决于弹窗类型）
            has_popup = self._check_conflict_popup()

            if has_popup:
                # 关闭弹窗
                self._close_popup()
                return True  # 冲突
            else:
                # 无冲突，恢复原始端口（避免改变配置）
                # 这里可能需要刷新页面或重新设置
                return False  # 可用

        except Exception as e:
            self.log_error(f"测试端口 {port} 时出错: {e}")
            return False

    def _check_conflict_popup(self) -> bool:
        """
        检测是否显示冲突弹窗

        具体实现取决于弹窗类型:
        - 如果是alert: 使用 switch_to.alert
        - 如果是自定义弹窗: 使用 find_element

        Returns:
            bool: True=有弹窗, False=无弹窗
        """
        # 方法1: 检测alert（待确认）
        try:
            alert = self.router_client.driver.switch_to.alert
            return True
        except:
            pass

        # 方法2: 检测自定义弹窗（待确认）
        if self.popup_xpath:
            try:
                popup = self.router_client.driver.find_element(
                    By.XPATH, self.popup_xpath
                )
                return popup.is_displayed()
            except:
                pass

        return False

    def _close_popup(self):
        """关闭冲突弹窗"""
        # 方法1: 关闭alert
        try:
            alert = self.router_client.driver.switch_to.alert
            alert.accept()
            return
        except:
            pass

        # 方法2: 关闭自定义弹窗（点击确定按钮）
        try:
            close_btn = self.router_client.driver.find_element(
                By.XPATH, "//button[contains(text(), '确定')]"
            )
            close_btn.click()
        except:
            pass

    def _compare_results(self):
        """对比实际结果和研发提供的列表"""
        self.log_info("="*70)
        self.log_info("结果对比分析")
        self.log_info("="*70)

        # 转换为集合便于对比
        provided_set = set(self.STATIC_PORTS_PROVIDED)
        actual_set = set(self.actual_conflict_ports)

        # 1. 遗漏的端口（实际冲突但列表没有）
        missing_ports = actual_set - provided_set

        # 2. 多余的端口（列表有但实际不冲突）
        extra_ports = provided_set - actual_set

        # 3. 正确的端口（列表有且实际冲突）
        correct_ports = provided_set & actual_set

        # 输出结果
        self.log_info(f"\n1. 研发提供的列表（{len(provided_set)}个端口）:")
        self.log_info(f"   {sorted(provided_set)}")

        self.log_info(f"\n2. 实际检测到的冲突端口（{len(actual_set)}个）:")
        self.log_info(f"   {sorted(actual_set)}")

        self.log_info(f"\n3. 列表正确的端口（{len(correct_ports)}个）:")
        self.log_info(f"   {sorted(correct_ports)}")

        if missing_ports:
            self.log_error(f"\n❌ 4. 列表遗漏的端口（{len(missing_ports)}个）:")
            self.log_error(f"   {sorted(missing_ports)}")
            self.log_error(f"   这些端口实际冲突，但研发列表中没有！")
        else:
            self.log_info(f"\n✅ 4. 列表无遗漏")

        if extra_ports:
            self.log_error(f"\n❌ 5. 列表多余的端口（{len(extra_ports)}个）:")
            self.log_error(f"   {sorted(extra_ports)}")
            self.log_error(f"   这些端口在列表中，但实际未冲突！")
        else:
            self.log_info(f"\n✅ 5. 列表无多余")

        # 准确率统计
        if len(provided_set) > 0:
            accuracy = (len(correct_ports) / len(provided_set)) * 100
            self.log_info(f"\n列表准确率: {accuracy:.1f}% ({len(correct_ports)}/{len(provided_set)})")

        # 完整性统计
        if len(actual_set) > 0:
            completeness = (len(correct_ports) / len(actual_set)) * 100
            self.log_info(f"列表完整性: {completeness:.1f}% ({len(correct_ports)}/{len(actual_set)})")

        self.log_info("="*70)

        # 存储结果用于验证
        self.missing_ports = missing_ports
        self.extra_ports = extra_ports
        self.correct_ports = correct_ports

    def verify(self):
        """验证测试结果"""
        self.log_info("\n验证测试结果...")

        # 断言：不能有遗漏
        if self.missing_ports:
            raise AssertionError(
                f"❌ 静态端口列表有遗漏！缺少{len(self.missing_ports)}个端口: "
                f"{sorted(self.missing_ports)}"
            )

        # 断言：不能有多余
        if self.extra_ports:
            raise AssertionError(
                f"❌ 静态端口列表有多余！{len(self.extra_ports)}个端口实际未占用: "
                f"{sorted(self.extra_ports)}"
            )

        # 全部正确
        self.log_info(f"✅ 静态端口列表验证通过！")
        self.log_info(f"   列表中的{len(self.STATIC_PORTS_PROVIDED)}个端口全部准确无误")

    def cleanup(self):
        """测试清理"""
        self.log_info("\n清理测试环境...")
        # 关闭浏览器等
        super().cleanup()
```

## ⚡ 性能优化建议

### 问题：遍历65535个端口耗时长

**预估时间**:
- 每个端口测试: 输入(0.5s) + 点击(0.5s) + 检测(0.5s) + 关闭弹窗(0.5s) = **2秒**
- 总耗时: 65535 × 2秒 = **131,070秒 = 36.4小时** ⚠️

### 优化方案1: 分段测试（推荐）

```python
# 只测试重点端口段
TEST_RANGES = [
    (1, 1024),       # 特权端口（必测）
    (1024, 10000),   # 常用端口
    (40000, 50000),  # 研发列表中的高端口段
    (55000, 56000),  # 研发列表中的高端口段
]

# 加上研发列表中的所有端口（确保必测）
MUST_TEST_PORTS = [22, 53, 1701, 9001, 68, 55832, 40404, 500, 4500, 123, 1, 58, 7574, 9993, 520]
```

**预估时间**: 约15000个端口 × 2秒 = **8.3小时**

### 优化方案2: 减少等待时间

```python
# 优化后每个端口耗时
每个端口测试: 输入(0.2s) + 点击(0.2s) + 检测(0.3s) + 关闭(0.2s) = 0.9秒

# 总耗时
65535 × 0.9秒 = 16.4小时（仍然很长）
```

### 优化方案3: 分批并行（高级，可选）

```python
# 使用多个浏览器实例并行测试不同端口段
# 4个线程，每个测试16384个端口
# 总耗时减少到约 9小时
```

### 推荐策略：组合优化

```python
1. 必测端口: 研发列表中的15个端口（2秒 × 15 = 30秒）
2. 特权端口: 1-1024（2秒 × 1024 = 34分钟）
3. 抽样测试: 1024-65535中每隔10个测试一个（6451个端口，3.6小时）

总耗时: 约4小时
```
